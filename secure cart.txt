<?php
include '../db.php';
session_start();

// Check if the user is logged in
if (!isset($_SESSION['user_id'])) {
    header('Location: login.php');
    exit;
}

$user_id = $_SESSION['user_id'];

// Generate a CSRF token
if (!isset($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

// Function to check if a cart item belongs to the current user
function isCartItemBelongsToUser($conn, $cart_id, $user_id) {
    $checkQuery = mysqli_query($conn, "SELECT * FROM cart WHERE id = '$cart_id' AND user_id = '$user_id'");
    return mysqli_num_rows($checkQuery) > 0;
}

// Delete a cart item
if (isset($_GET['delete']) && isset($_GET['csrf_token']) && $_GET['csrf_token'] === $_SESSION['csrf_token']) {
    $cart_id = $_GET['delete'];
    if (isCartItemBelongsToUser($conn, $cart_id, $user_id)) {
        mysqli_query($conn, "DELETE FROM cart WHERE id = '$cart_id'");
        echo "<script>alert('Item removed!'); window.location='cart.php';</script>";
    } else {
        echo "<script>alert('You do not have permission to delete this item!'); window.location='cart.php';</script>";
    }
}

// Update the cart quantity
if (isset($_GET['update']) && isset($_GET['quantity']) && isset($_GET['csrf_token']) && $_GET['csrf_token'] === $_SESSION['csrf_token']) {
    $cart_id = $_GET['update'];
    $quantity = $_GET['quantity'];
    // Validate that the quantity is a positive integer
    if (filter_var($quantity, FILTER_VALIDATE_INT, array("options" => array("min_range" => 1))) === false) {
        echo "<script>alert('Invalid quantity!'); window.location='cart.php';</script>";
    } else {
        if (isCartItemBelongsToUser($conn, $cart_id, $user_id)) {
            mysqli_query($conn, "UPDATE cart SET quantity = '$quantity' WHERE id = '$cart_id'");
            echo "<script>alert('Quantity updated!'); window.location='cart.php';</script>";
        } else {
            echo "<script>alert('You do not have permission to update this item!'); window.location='cart.php';</script>";
        }
    }
}

// Send a message to the seller
if (isset($_POST['message']) && isset($_POST['csrf_token']) && $_POST['csrf_token'] === $_SESSION['csrf_token']) {
    $message = $_POST['message'];
    // First, remove extra whitespace
    $message = trim($message);
    // Then, remove non - allowed characters
    $message = preg_replace('/[^a-zA-Z0-9\s.,]/', '', $message);
    // Sanitize to prevent XSS attacks
    $message = htmlspecialchars($message, ENT_QUOTES, 'UTF-8');
    // Prevent SQL injection
    $message = mysqli_real_escape_string($conn, $message);
    if (empty($message)) {
        echo "<script>alert('Please enter a valid message.'); window.location='cart.php';</script>";
        exit;
    }
    mysqli_query($conn, "INSERT INTO seller_messages (user_id, message) VALUES ('$user_id', '$message')");
    echo "<script>alert('Message sent!'); window.location='cart.php';</script>";
}

// Fetch the user's cart
$cart_items = mysqli_query($conn, "SELECT cart.*, books.title, books.price 
                                   FROM cart 
                                   JOIN books ON cart.book_id = books.id 
                                   WHERE cart.user_id = '$user_id'");

// Calculate the total price
$total = 0;
$itemCount = 0;
$cartData = [];
while ($row = mysqli_fetch_assoc($cart_items)) {
    $subtotal = $row['price'] * $row['quantity'];
    $total += $subtotal;
    $itemCount++;
    $cartData[] = $row;
}
// Reset the result pointer to the start
mysqli_data_seek($cart_items, 0);

    <script>
        const cartData = <?php echo json_encode($cartData); ?>;
        const csrfToken = '<?php echo $_SESSION['csrf_token']; ?>';

        function changeQuantity(index, delta) {
            const input = document.querySelector(`input[onchange*='updateQuantity(${index},']`);
            let quantity = parseInt(input.value);
            quantity += delta;
            if (quantity < 1) {
                alert('Please choose a minimum of 1 quantity');
                return;
            }
            input.value = quantity;
            updateQuantity(index, quantity);
        }

        function updateQuantity(index, quantity) {
            const price = cartData[index].price;
            const subtotal = price * quantity;
            const subtotalElement = document.getElementById(`subtotal-${index}`);
            subtotalElement.textContent = subtotal.toFixed(2);

            let newTotal = 0;
            const quantityInputs = document.querySelectorAll('.quantity-controls input');
            quantityInputs.forEach((input, i) => {
                const itemQuantity = parseInt(input.value);
                const itemPrice = cartData[i].price;
                newTotal += itemPrice * itemQuantity;
            });

            const itemTotalElement = document.getElementById('item-total');
            const totalPriceElement = document.getElementById('total-price');
            itemTotalElement.textContent = `€ ${newTotal.toFixed(2)}`;
            totalPriceElement.textContent = `€ ${newTotal.toFixed(2)}`;

            // Send a request to update the database
            const cartId = cartData[index].id;
            window.location.href = `cart.php?update=${cartId}&quantity=${quantity}&csrf_token=${csrfToken}`;
        }

        function deleteItem(itemId) {
            if (confirm('Are you sure you want to remove this item?')) {
                window.location.href = `cart.php?delete=${itemId}&csrf_token=${csrfToken}`;
            }
        }

        function validateMessage(textarea) {
            const value = textarea.value;
            const valid = /^[a-zA-Z0-9\s.,]+$/.test(value);
            if (!valid) {
                alert('Please only enter words, numbers, full stops (.), or commas (,).');
                textarea.value = textarea.value.replace(/[^a-zA-Z0-9\s.,]/g, '');
            }
        }
    </script>
